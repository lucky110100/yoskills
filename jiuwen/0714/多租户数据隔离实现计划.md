# 多租户数据隔离 实现计划

> **面向 AI 代理的工作者：** 必需子技能：使用 superpowers:subagent-driven-development（推荐）或 superpowers:executing-plans 逐任务实现此计划。步骤使用复选框（`- [ ]`）语法来跟踪进度。

**目标：** 为 agent-core-java 实现 tenant_id 驱动的多租户数据隔离，覆盖 Workspace、SysOperation、Checkpointer、Skill 四大隔离维度。

**架构：** 无状态动态隔离——单 DeepAgent 实例共享，每次 invoke 时通过 TenantExecutionContext（ThreadLocal）动态注入租户隔离资源。隔离逻辑全部在 DeepAgent 层完成，ReActAgent 不感知租户。

**技术栈：** Java 17、Lombok、JUnit 5、现有 agent-core-java 框架

package com.openjiuwen.core.multitenant;

import java.io.IOException;
import java.nio.file.Path;

public final class TenantPathValidator {

    private TenantPathValidator() {
    }

    public static void validate(String tenantId, Path requestedPath) {
        if (tenantId == null || tenantId.isBlank()) {
            return;
        }
        Path tenantRoot = resolveTenantRoot(tenantId).normalize().toAbsolutePath();
        Path normalized = requestedPath.normalize().toAbsolutePath();
        if (!normalized.startsWith(tenantRoot)) {
            throw new SecurityException("Cross-tenant access denied: " + requestedPath);
        }
        try {
            Path resolved = normalized.toRealPath();
            if (!resolved.startsWith(tenantRoot)) {
                throw new SecurityException("Cross-tenant symlink access denied: " + requestedPath);
            }
        } catch (IOException e) {
            throw new SecurityException("Path validation failed: " + requestedPath, e);
        }
    }

    static Path resolveTenantRoot(String tenantId) {
        return Path.of("tenants").resolve(tenantId);
    }
}

package com.openjiuwen.core.multitenant;

import com.openjiuwen.harness.workspace.Workspace;

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;

public class TenantWorkspaceManager {

    private final Path baseDataDir;
    private final Path sharedDir;

    public TenantWorkspaceManager(Path baseDataDir) {
        this.baseDataDir = baseDataDir != null ? baseDataDir : Path.of(".");
        this.sharedDir = this.baseDataDir.resolve("shared");
    }

    public Workspace resolveWorkspace(String tenantId) {
        Path root = tenantId != null && !tenantId.isBlank()
            ? baseDataDir.resolve("tenants").resolve(tenantId).resolve("workspace")
            : sharedDir.resolve("workspace");
        return Workspace.builder()
            .rootPath(root.toString())
            .language("cn")
            .build();
    }

    public Path resolveTmpDir(String tenantId) {
        return resolveWorkspace(tenantId).root().resolve("tmp");
    }

    public void initializeTenant(String tenantId) {
        if (tenantId == null || tenantId.isBlank()) {
            return;
        }
        Path tenantRoot = baseDataDir.resolve("tenants").resolve(tenantId);
        try {
            Files.createDirectories(tenantRoot.resolve("workspace").resolve("tmp"));
            Files.createDirectories(tenantRoot.resolve("workspace").resolve("skills"));
            Files.createDirectories(tenantRoot.resolve("checkpoints"));
        } catch (IOException e) {
            throw new IllegalStateException("Failed to initialize tenant directory: " + tenantId, e);
        }
    }
}

package com.openjiuwen.core.multitenant;

import com.openjiuwen.core.sysop.SysOperation;
import com.openjiuwen.core.sysop.OperationMode;
import com.openjiuwen.core.sysop.SysOperationCard;
import com.openjiuwen.core.sysop.config.LocalWorkConfig;
import com.openjiuwen.harness.workspace.Workspace;

import java.nio.file.Path;
import java.util.List;

public class TenantSysOpFactory {

    private final TenantWorkspaceManager workspaceManager;

    public TenantSysOpFactory(TenantWorkspaceManager workspaceManager) {
        this.workspaceManager = workspaceManager;
    }

    public SysOperation createForTenant(String tenantId) {
        Workspace tenantWorkspace = workspaceManager.resolveWorkspace(tenantId);
        Path tenantRoot = tenantWorkspace.root();

        LocalWorkConfig config = LocalWorkConfig.builder()
            .workDir(tenantRoot.toString())
            .sandboxRoot(List.of(tenantRoot.toString()))
            .restrictToSandbox(true)
            .build();

        SysOperationCard card = SysOperationCard.builder()
            .id("sys_op_" + (tenantId != null ? tenantId : "shared"))
            .mode(OperationMode.LOCAL)
            .workConfig(config)
            .build();

        return new SysOperation(card);
    }
}

package com.openjiuwen.core.multitenant;

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.attribute.BasicFileAttributes;
import java.time.Duration;
import java.time.Instant;

public class TenantResourceQuota {

    private final TenantWorkspaceManager workspaceManager;
    private final long maxTmpSizeBytes;
    private final Duration tmpTtl;

    public TenantResourceQuota(TenantWorkspaceManager workspaceManager,
                                long maxTmpSizeBytes, Duration tmpTtl) {
        this.workspaceManager = workspaceManager;
        this.maxTmpSizeBytes = maxTmpSizeBytes;
        this.tmpTtl = tmpTtl;
    }

    public void cleanupExpiredTmp(String tenantId) {
        Path tmpDir = workspaceManager.resolveTmpDir(tenantId);
        if (!Files.isDirectory(tmpDir)) {
            return;
        }
        Instant cutoff = Instant.now().minus(tmpTtl);
        try {
            Files.walk(tmpDir)
                .filter(Files::isRegularFile)
                .filter(file -> {
                    try {
                        BasicFileAttributes attrs = Files.readAttributes(file, BasicFileAttributes.class);
                        return attrs.lastModifiedTime().toInstant().isBefore(cutoff);
                    } catch (IOException e) {
                        return false;
                    }
                })
                .forEach(file -> {
                    try {
                        Files.deleteIfExists(file);
                    } catch (IOException ignored) {
                    }
                });
        } catch (IOException ignored) {
        }
    }
}
